```ts
// src/services/http/httpClient.ts
/**
 * @description HTTP Client global da aplicação.
 * Injeta token JWT automaticamente em todas as requisições.
 * Parseia ApiResponse<T> do backend e converte para Result<T> internamente.
 * A camada de services nunca lida com ApiResponse diretamente.
 * @since 2024-01-15
 */
import { httpInterceptors } from './httpInterceptors';
import { httpErrorHandler } from './httpErrorHandler';
import type { HttpResult, RequestConfig } from './models/http.types';
import type { ApiResponse } from '@/models/apiResponse.types';

const BASE_URL = import.meta.env.VITE_API_URL;

async function request<T>(method: string, path: string, config?: RequestConfig): Promise<HttpResult<T>> {
  const headers = await httpInterceptors.buildHeaders();
  const url = `${BASE_URL}${path}${config?.params ? `?${new URLSearchParams(config.params as Record<string, string>)}` : ''}`;

  try {
    const response = await fetch(url, {
      method,
      headers,
      body: config?.body ? JSON.stringify(config.body) : undefined,
    });

    const apiResponse: ApiResponse<T> = await response.json();

    if (!response.ok || apiResponse.severity === 'error') {
      return httpErrorHandler.handleApiError(apiResponse);
    }

    return { ok: true, data: apiResponse.data };
  } catch (err) {
    return httpErrorHandler.handleNetworkError(err);
  }
}

export const httpClient = {
  get: <T>(path: string, config?: RequestConfig) => request<T>('GET', path, config),
  post: <T>(path: string, body: unknown, config?: RequestConfig) => request<T>('POST', path, { ...config, body }),
  put: <T>(path: string, body: unknown, config?: RequestConfig) => request<T>('PUT', path, { ...config, body }),
  patch: <T>(path: string, body: unknown, config?: RequestConfig) => request<T>('PATCH', path, { ...config, body }),
  delete: <T>(path: string, config?: RequestConfig) => request<T>('DELETE', path, config),
};
```

```ts
// src/services/http/httpInterceptors.ts
/**
 * @description Interceptors HTTP — injeção de token JWT e headers padrão.
 */
import { authStorage } from '@/services/auth/authStorage';

export const httpInterceptors = {
  async buildHeaders(): Promise<HeadersInit> {
    const token = authStorage.getToken();
    return {
      'Content-Type': 'application/json',
      'Accept': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    };
  },
};
```

```ts
// src/services/http/httpDemoStrategy.ts
/**
 * @description Strategy HTTP para o modo demonstração.
 * Intercepta toda requisição e retorna dados mockados com delay simulado.
 * Permite que a aplicação funcione 100% sem API ativa — nem localmente.
 * O módulo deve ter seu próprio /demo/factory.ts que implementa as respostas.
 */
import { simulateDelay } from '@/utils/demo.util';
import type { HttpResult } from './models/http.types';

type DemoHandlerMap = Map<string, (body?: unknown) => Promise<HttpResult<unknown>>>;

const demoHandlers: DemoHandlerMap = new Map();

export const httpDemoStrategy = {
  /**
   * Registra um handler mockado para um endpoint específico.
   * Chamado pelos factories de cada módulo durante a inicialização.
   */
  register(method: string, path: string, handler: (body?: unknown) => Promise<HttpResult<unknown>>): void {
    demoHandlers.set(`${method}:${path}`, handler);
  },

  /**
   * Resolve a requisição com o handler mockado correspondente.
   * Se não houver handler registrado, retorna erro de "endpoint não encontrado".
   */
  async resolve<T>(method: string, path: string, body?: unknown): Promise<HttpResult<T>> {
    await simulateDelay(300, 800);
    const key = `${method}:${path}`;
    const handler = demoHandlers.get(key);
    if (!handler) {
      return { ok: false, error: `[Demo] Endpoint não mapeado: ${key}`, status: 404 };
    }
    return handler(body) as Promise<HttpResult<T>>;
  },
};
```

```ts
// src/services/http/httpErrorHandler.ts
/**
 * @description Converte ApiResponse de erro para Result<never>.
 * Emite eventos no EventBus para exibição de toasts.
 */
import { eventBus } from '@/services/eventBus/eventBus';
import type { HttpResult } from './models/http.types';
import type { ApiResponse } from '@/models/apiResponse.types';

export const httpErrorHandler = {
  handleApiError<T>(apiResponse: ApiResponse<T>): HttpResult<never> {
    if (apiResponse.status === 401) {
      eventBus.emit('auth:session:expired', {});
    }
    eventBus.emit('ui:toast:show', { type: 'error', message: apiResponse.message });
    return { ok: false, error: apiResponse.message, status: apiResponse.status };
  },

  handleNetworkError(err: unknown): HttpResult<never> {
    const message = 'Falha de conexão. Verifique sua internet.';
    eventBus.emit('ui:toast:show', { type: 'error', message });
    return { ok: false, error: message, status: 0 };
  },
};
```
