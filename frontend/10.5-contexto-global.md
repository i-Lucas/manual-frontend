# 10.5 Feature Global — Contexto Global Persistido

## Estrutura

```
src/contexts/
├── AppContext.tsx          ← Provider raiz que agrega sub-contexts
├── AuthContext/
│   ├── AuthContext.tsx
│   ├── AuthProvider.tsx
│   └── useAuth.ts
└── CompanyContext/
    ├── CompanyContext.tsx
    ├── CompanyProvider.tsx
    └── useCompany.ts
```

## Ordenação de providers

**Providers que mudam com menos frequência ficam por fora; providers que mudam mais ficam por dentro.** Isso minimiza re-renders: quando um provider interno atualiza, apenas seus filhos re-renderizam — não os siblings dos providers externos.

```tsx
// src/contexts/AppContext.tsx — agrega todos os providers na ordem correta
export function AppContext({ children }: { children: React.ReactNode }) {
  return (
    // Raramente muda — por fora
    <AuthProvider>
      <CompanyProvider>
        {/* Muda com mais frequência — por dentro */}
        <ToastProvider>
          {children}
        </ToastProvider>
      </CompanyProvider>
    </AuthProvider>
  );
}
```

```tsx
// src/main.tsx
ReactDOM.createRoot(document.getElementById('root')!).render(
  <StrictMode>
    <AppContext>
      <AppRouter />
    </AppContext>
  </StrictMode>
);
```

> O `ToastContainer` — componente que escuta `ui:toast:show` e renderiza os toasts — deve ser instanciado dentro do layout global (ex: `AppRouter` ou um layout wrapper), não dentro de módulos individuais.

## AuthProvider

```tsx
// src/contexts/AuthContext/AuthProvider.tsx
/**
 * @description Provider de autenticação. Persiste sessão em localStorage.
 * Ouve evento 'auth:session:expired' para logout automático.
 */
import { useState, useEffect, useCallback, useMemo } from 'react';
import { AuthContext } from './AuthContext';
import { authStorage } from '@/services/auth/authStorage';
import { eventBus } from '@/services/eventBus/eventBus';
import type { AuthUser } from '@/models/auth.types';

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<AuthUser | null>(() => authStorage.getUser());

  const logout = useCallback(() => {
    authStorage.clear();
    setUser(null);
  }, []);

  const login = useCallback((u: AuthUser, token: string) => {
    authStorage.save(u, token);
    setUser(u);
  }, []);

  useEffect(() => {
    return eventBus.on('auth:session:expired', () => logout());
  }, [logout]);

  const value = useMemo(() => ({ user, login, logout, isAuthenticated: !!user }), [user, login, logout]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}
```
